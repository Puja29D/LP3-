import pandas as pd
import matplotlib.pyplot as plt

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# 1. Load Dataset

df = pd.read_csv("sales_data_sample.csv", encoding="ISO-8859-1")

print("First 5 records:")
print(df.head())

print("\nDataset Shape:")
print(df.shape)

# 2. Select Numerical Features

# We use sales-related numerical attributes
features = [
    "QUANTITYORDERED",
    "PRICEEACH",
    "SALES"
]

X = df[features]

# Check for missing values
print("\nMissing values:")
print(X.isnull().sum())

# 3. Standardize the Data

scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# 4. Elbow Method

inertia = []

for k in range(1, 11):
    kmeans = KMeans(
        n_clusters=k,
        random_state=42,
        n_init=10
    )

    kmeans.fit(X_scaled)
    inertia.append(kmeans.inertia_)

# Plot Elbow Curve
plt.figure(figsize=(8, 5))
plt.plot(range(1, 11), inertia, marker="o")
plt.xlabel("Number of Clusters (K)")
plt.ylabel("Inertia")
plt.title("Elbow Method")
plt.xticks(range(1, 11))
plt.grid(True)
plt.show()


# 5. Apply K-Means

# Select K based on the elbow point
k = 3

kmeans = KMeans(
    n_clusters=k,
    random_state=42,
    n_init=10
)

df["Cluster"] = kmeans.fit_predict(X_scaled)

# 6. Display Cluster Information

print("\nCluster Counts:")
print(df["Cluster"].value_counts().sort_index())

print("\nCluster Centers:")
centers = scaler.inverse_transform(kmeans.cluster_centers_)

cluster_centers = pd.DataFrame(
    centers,
    columns=features
)

print(cluster_centers)

# 7. Display Sample Clustered Data

print("\nClustered Data:")
print(df[features + ["Cluster"]].head(20))

# 8. Visualize Clusters

plt.figure(figsize=(8, 6))

plt.scatter(
    df["QUANTITYORDERED"],
    df["SALES"],
    c=df["Cluster"],
    cmap="viridis",
    s=50
)

plt.xlabel("Quantity Ordered")
plt.ylabel("Sales")
plt.title("K-Means Clustering of Sales Data")
plt.colorbar(label="Cluster")
plt.grid(True)
plt.show()











Q1. What is clustering?

Clustering is an unsupervised machine learning technique that groups similar data points together.

Q2. What is K-Means clustering?

K-Means divides data into K clusters based on the distance from cluster centers called centroids.

Q3. What is hierarchical clustering?

It is an unsupervised learning algorithm that creates a hierarchy of clusters in a tree-like structure.

Q4. What is the difference between supervised and unsupervised learning?
- Supervised learning uses labelled data.
- Unsupervised learning uses unlabelled data.
Q5. What is the Elbow Method?

The Elbow Method helps determine the optimal number of clusters by plotting WCSS against the number of clusters.

Q6. What is WCSS?

WCSS stands for Within-Cluster Sum of Squares. It measures the sum of squared distances between data points and their cluster centroids.

Q7. Why do we use the Elbow Method?

To find the optimal number of clusters.

Q8. What is a centroid?

A centroid is the center point of a cluster.

Q9. Why do we use StandardScaler?

To bring numerical features to a similar scale so that large-valued features do not dominate distance calculations.

Q10. What is a dendrogram?

A dendrogram is a tree-like diagram that represents how clusters are merged in hierarchical clustering.

Q11. What are the two types of hierarchical clustering?
1. Agglomerative (bottom-up).
2. Divisive (top-down).
Q12. What is Agglomerative clustering?

It starts with each data point as a separate cluster and repeatedly merges the closest clusters.

Q13. What is the Ward method?

Ward's method merges clusters in a way that minimizes the increase in within-cluster variance.

Q14. What is the difference between K-Means and Hierarchical clustering?

Q15. What is the use of clustering in sales data?

It helps group customers or sales records based on purchasing quantity and sales value.
