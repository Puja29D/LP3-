import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import r2_score, mean_squared_error, mean_absolute_error


# 1. LOAD DATASET

df = pd.read_csv("uber.csv")

print("Original Shape:", df.shape)
print(df.head())


# 2. PREPROCESSING

# Remove unnecessary columns
df.drop(["Unnamed: 0", "key"], axis=1, inplace=True)

# Remove missing values
df.dropna(inplace=True)

# Convert datetime
df["pickup_datetime"] = pd.to_datetime(df["pickup_datetime"])

# Extract date/time features
df["year"] = df["pickup_datetime"].dt.year
df["month"] = df["pickup_datetime"].dt.month
df["day"] = df["pickup_datetime"].dt.day
df["hour"] = df["pickup_datetime"].dt.hour
df["day_of_week"] = df["pickup_datetime"].dt.dayofweek

# Remove original datetime
df.drop("pickup_datetime", axis=1, inplace=True)



# 3. REMOVE INVALID VALUES

df = df[
    (df["pickup_longitude"] >= -74.3) &
    (df["pickup_longitude"] <= -73.7) &
    (df["dropoff_longitude"] >= -74.3) &
    (df["dropoff_longitude"] <= -73.7) &
    (df["pickup_latitude"] >= 40.4) &
    (df["pickup_latitude"] <= 41.0) &
    (df["dropoff_latitude"] >= 40.4) &
    (df["dropoff_latitude"] <= 41.0)
]

df = df[
    (df["fare_amount"] > 0) &
    (df["passenger_count"] > 0) &
    (df["passenger_count"] <= 6)
]


# 4. CALCULATE DISTANCE

def haversine_distance(lat1, lon1, lat2, lon2):

    R = 6371

    lat1 = np.radians(lat1)
    lon1 = np.radians(lon1)

    lat2 = np.radians(lat2)
    lon2 = np.radians(lon2)

    dlat = lat2 - lat1
    dlon = lon2 - lon1

    a = (
        np.sin(dlat / 2) ** 2
        + np.cos(lat1)
        * np.cos(lat2)
        * np.sin(dlon / 2) ** 2
    )

    c = 2 * np.arcsin(np.sqrt(a))

    return R * c


df["distance_km"] = haversine_distance(
    df["pickup_latitude"],
    df["pickup_longitude"],
    df["dropoff_latitude"],
    df["dropoff_longitude"]
)

# 5. OUTLIER REMOVAL

def remove_outliers_iqr(data, column):

    Q1 = data[column].quantile(0.25)
    Q3 = data[column].quantile(0.75)

    IQR = Q3 - Q1

    lower = Q1 - 1.5 * IQR
    upper = Q3 + 1.5 * IQR

    return data[
        (data[column] >= lower) &
        (data[column] <= upper)
    ]


df = remove_outliers_iqr(df, "fare_amount")
df = remove_outliers_iqr(df, "distance_km")


print("\nShape after preprocessing:", df.shape)


# 6. CORRELATION

plt.figure(figsize=(12, 8))

sns.heatmap(
    df.corr(),
    annot=True,
    cmap="coolwarm",
    fmt=".2f"
)

plt.title("Correlation Matrix")
plt.show()


# 7. FEATURES AND TARGET

features = [
    "pickup_longitude",
    "pickup_latitude",
    "dropoff_longitude",
    "dropoff_latitude",
    "passenger_count",
    "year",
    "month",
    "day",
    "hour",
    "day_of_week",
    "distance_km"
]

X = df[features]
y = df["fare_amount"]

# 8. TRAIN TEST SPLIT

X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=42
)


# 9. LINEAR REGRESSION

linear_model = LinearRegression()

linear_model.fit(X_train, y_train)

y_pred_linear = linear_model.predict(X_test)


linear_r2 = r2_score(
    y_test,
    y_pred_linear
)

linear_rmse = np.sqrt(
    mean_squared_error(
        y_test,
        y_pred_linear
    )
)

linear_mae = mean_absolute_error(
    y_test,
    y_pred_linear
)


# 10. RANDOM FOREST

rf_model = RandomForestRegressor(
    n_estimators=100,
    random_state=42,
    n_jobs=-1
)

rf_model.fit(X_train, y_train)

y_pred_rf = rf_model.predict(X_test)


rf_r2 = r2_score(
    y_test,
    y_pred_rf
)

rf_rmse = np.sqrt(
    mean_squared_error(
        y_test,
        y_pred_rf
    )
)

rf_mae = mean_absolute_error(
    y_test,
    y_pred_rf
)
# 11. MODEL COMPARISON

results = pd.DataFrame({

    "Model": [
        "Linear Regression",
        "Random Forest Regression"
    ],

    "R2 Score": [
        linear_r2,
        rf_r2
    ],

    "RMSE": [
        linear_rmse,
        rf_rmse
    ],

    "MAE": [
        linear_mae,
        rf_mae
    ]
})


print("\n================ MODEL COMPARISON ================")
print(results)


# 12. FEATURE IMPORTANCE

importance = pd.DataFrame({

    "Feature": X.columns,

    "Importance":
        rf_model.feature_importances_
})

importance = importance.sort_values(
    by="Importance",
    ascending=False
)

print("\nRandom Forest Feature Importance:")
print(importance)

Part 7: Model evaluation metrics
62. Why do we evaluate a Machine Learning model?

We evaluate a model to measure how accurately it predicts unseen data and to compare different algorithms.

63. What is R² Score?

R² Score, also called the coefficient of determination, measures how much variation in the target variable is explained by the model.
\[
R^2=1-\frac{SS_{res}}{SS_{tot}}
\]

- 1 = Perfect prediction.
- 0 = No improvement over predicting the test-set mean.
- Negative = Worse than predicting the test-set mean.
64. What is MAE?

MAE stands for Mean Absolute Error. It calculates the average absolute difference between actual and predicted values.
\[
MAE=\frac{1}{n}\sum|y-\hat y|
\]

65. What is MSE?

MSE stands for Mean Squared Error. It calculates the average squared difference between actual and predicted values.
\[
MSE=\frac{1}{n}\sum(y-\hat y)^2
\]

66. What is RMSE?

RMSE stands for Root Mean Squared Error. It is the square root of MSE.
\[
RMSE=\sqrt{MSE}
\]

67. Which evaluation metric is better: MAE or RMSE?

Both are useful.
- MAE is easier to interpret and less sensitive to large errors.
- RMSE penalizes larger errors more heavily.
68. Which R² score is better?

Generally, a higher R² score indicates better model performance on the same test dataset.

69. Which RMSE value is better?

A lower RMSE value indicates that the model's predictions are closer to actual values on average in terms of squared error.

70. Why do we use multiple evaluation metrics?

Because one metric cannot explain every aspect of model performance. Using R², MAE and RMSE provides a more complete understanding.

71. Can R² be negative?

Yes. It means the model performs worse than a baseline that predicts the mean of the actual target values.

72. What is prediction error?

Prediction error is the difference between the actual value and predicted value.
\[
Error=Actual-Predicted
\]
